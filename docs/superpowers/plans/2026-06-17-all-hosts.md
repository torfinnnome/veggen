```python
def get_all_hosts():
    """Fetch all network hosts from static DHCP and dynamic leases, deduplicated by MAC."""
    result = {}

    # 1. Static hosts from uci
    dhcp_output = run_ssh_command("sudo uci show dhcp")
    if dhcp_output:
        hosts = {}
        for line in dhcp_output.splitlines():
            match = re.match(r"dhcp\.(@host\[\d+\]|[a-zA-Z0-9_-]+)\.(\w+)='?([^']*)'?", line)
            if match:
                section, key, value = match.groups()
                if section not in hosts:
                    hosts[section] = {}
                hosts[section][key] = value
        for section, data in hosts.items():
            mac = data.get("mac", "").lower()
            name = data.get("name", "")
            ip = data.get("ip", "")
            if mac and ip:
                result[mac] = {"name": name, "ip": ip, "mac": mac}

    # 2. Dynamic leases from /tmp/dhcp.leases
    leases_output = run_ssh_command("cat /tmp/dhcp.leases 2>/dev/null")
    if leases_output:
        for line in leases_output.splitlines():
            parts = line.split()
            if len(parts) >= 4:
                # Format: expiry ip mac name hostname
                ip, mac, name = parts[1], parts[2].lower(), parts[3]
                if mac and ip and mac not in result:
                    result[mac] = {"name": name if name else "unknown", "ip": ip, "mac": mac}

    return list(result.values())
```

```javascript
async function fetchAllHosts() {
    try {
        var res = await fetch('/api/all-hosts');
        var hosts = await res.json();
        var tbody = document.getElementById('allhosts-tbody');
        tbody.innerHTML = '';
        if (hosts.length === 0) {
            var row = document.createElement('tr');
            var cell = document.createElement('td');
            cell.setAttribute('colspan', '4');
            cell.textContent = 'No hosts found.';
            row.appendChild(cell);
            tbody.appendChild(row);
            return;
        }
        hosts.forEach(function(host) {
            var tr = document.createElement('tr');
            tr.className = 'device-row';
            var nameTd = document.createElement('td');
            nameTd.textContent = host.name;
            tr.appendChild(nameTd);
            var ipTd = document.createElement('td');
            ipTd.textContent = host.ip;
            tr.appendChild(ipTd);
            var macTd = document.createElement('td');
            macTd.textContent = host.mac;
            tr.appendChild(macTd);
            var nowTd = document.createElement('td');
            nowTd.className = 'now-cell';
            nowTd.innerHTML = '<span style="color: #1565c0;">\u2193 <span class="down-speed" data-mac="' + host.mac + '">\u2014</span></span><br><span style="color: #c62828;">\u2191 <span class="up-speed" data-mac="' + host.mac + '">\u2014</span></span>';
            tr.appendChild(nowTd);
            tbody.appendChild(tr);
            _detailCache[host.mac] = _detailCache[host.mac] || null;
        });
    } catch (e) {
        console.error('Error fetching all hosts:', e);
    }
}
```

```javascript
function toggleDetail(dev) {
    var mac = dev.mac;
    var det = document.querySelector('.detail-row[data-mac="' + mac + '"]');
    if (!det || det.style.display === 'none') {
        if (!det) {
            var speedEl = document.querySelector('.down-speed[data-mac="' + mac + '"]');
            if (!speedEl) return;
            var parentTr = speedEl.closest('tr');
            det = _createDetailRow(mac);
            parentTr.parentNode.insertBefore(det, parentTr.nextSibling);
        }
        det.style.display = '';
        (async function () {
            var period = 'day';
            var data = _detailCache[mac];
            if (!data) data = await _fetchHistory(mac, period);
            if (data && data.period) period = data.period;
            _renderDetail(mac, data || {}, period);
        })();
    } else {
        det.style.display = 'none';
    }
}
```

The key change: `parentTr = speedEl.closest('tr')` instead of `speedEl.closest('tr.device-row')` — this works for both managed and all-hosts rows.

```javascript
var _lastTrafficSummary = {};

async function _pollTrafficSummary() {
    try {
        var res = await fetch('/api/traffic/summary');
        var data = await res.json();
        _lastTrafficSummary = data;
        for (var mac in data) {
            var speeds = data[mac];
            var els = document.querySelectorAll('[data-mac="' + mac + '"]');
            els.forEach(function (el) {
                if (el.classList.contains('down-speed')) el.textContent = speeds.down_text || '\u2014';
                else if (el.classList.contains('up-speed')) el.textContent = speeds.up_text || '\u2014';
            });
        }
        _updateAllHostsVisibility();
    } catch (e) { /* non-critical */ }
}

function _updateAllHostsVisibility() {
    var tbody = document.getElementById('allhosts-tbody');
    if (!tbody) return;
    var rows = tbody.querySelectorAll('tr');
    rows.forEach(function(row) {
        var macCell = row.querySelector('td:nth-child(3)');
        if (!macCell) return;
        var mac = macCell.textContent;
        var nowCell = row.querySelector('td:nth-child(4)');
        if (!nowCell) return;
        if (_lastTrafficSummary[mac]) {
            if (!nowCell.querySelector('.down-speed')) {
                nowCell.innerHTML = '<span style="color: #1565c0; cursor:pointer;" class="speed-click" data-mac="' + mac + '">\u2193 <span class="down-speed" data-mac="' + mac + '">\u2014</span></span><br><span style="color: #c62828; cursor:pointer;" class="speed-click" data-mac="' + mac + '">\u2191 <span class="up-speed" data-mac="' + mac + '">\u2014</span></span>';
            }
        } else {
            if (nowCell.querySelector('.down-speed')) {
                nowCell.innerHTML = '';
            }
        }
    });
    setupSpeedClicks();
}
```
